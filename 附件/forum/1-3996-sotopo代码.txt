# To return an action, assign: action = {...}
sales_name=record.name
tolist=[]
for line in record.order_line:
    cpid=line.product_id
    order_line_copy=(0, 0, {
                'name': line.name,
                'product_id': line.product_id.id,
                'product_qty': line.product_uom_qty,
                'price_unit': line.price_unit,
                'taxes_id': line.tax_id.ids,
                #'product_uom': 1,
                'product_uom': line.product_uom.id

            })
    tolist.append(order_line_copy)
    
purchase_line = env['purchase.order'].search([
            ('origin', '=', sales_name),
        ], limit=1)
        
#raise UserError("测试%s" % purchase_line)        
        
if purchase_line:
    purchase_group_id=purchase_line.purchase_group_id.id
    result = env['ir.actions.act_window']._for_xml_id('purchase.purchase_rfq')
    result['context'] = {}
    if purchase_group_id:
        result['domain'] = ['|', ('origin', '=', sales_name), ('purchase_group_id', '=', purchase_group_id)]
    else:
        result['domain'] = [('origin', '=', sales_name)]
    action = result
else:
    result = env['ir.actions.actions']._for_xml_id('purchase.action_rfq_form')
    result.update({
                'view_mode': 'form',
                'view_id': env.ref('purchase.purchase_order_form').id,
                'views': [(env.ref('purchase.purchase_order_form').id, 'form')],
                'context': {'default_origin': sales_name,'default_order_line': tolist},
            })
    action = result
