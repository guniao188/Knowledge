## 补充/答案 2

在odoo中可以使用odoorpc进行接口操作，具体案例：不通过odoo系统操作创建一个销售订单

首先安装odoorpc（Python环境）： pip install odoorpc

说明：创建外键一对多时使用：(0, 0, {value})；外键多对多：(6, 0, [ids])

以下为操作案例输出结果：

![[1-370-b9fd0145.png]]

![[1-370-bff128ef.png]]

代码如下：

```python
    import odoorpc

    ip = '127.0.0.1'  # ip地址
    port = 8069       # 端口
    db_name = 'odoo_test'  # 数据库名称
    user_name = 'admin'    # 登录用户名
    password = 'admin'     # 登录密码

    class GetOdooData():

        def __init__(self):
            self.ip = ip
            self.port = port
            self.db_name = db_name
            self.user_name = user_name
            self.password = password

        def sign_odoo(self):
            odoo = odoorpc.ODOO(self.ip, port=self.port)
            odoo.login(self.db_name, self.user_name, self.password)
            return odoo

        def get_odoo_data(self):
            cls = self.sign_odoo()
            product_ids = cls.env['product.template'].search([('name', '=', '成品A')])  # 这里search出来的是一个id列表
            products = cls.env['product.template'].browse(product_ids)       # 这里才变为对象
            return products

        def create_sale_order(self):
            cls = self.sign_odoo()
            order_line_values = [
                {'customer_lead': 1.00,
                 'name': '订单明细1',
                 'price_unit': 10.0,
                 'product_uom_qty': 2.0,
                 'product_id': cls.env['product.product'].search([('product_tmpl_id', '=', self.get_odoo_data().id)])[0]},
                {'customer_lead': 2.00,
                 'name': '订单明细2',
                 'price_unit': 12.0,
                 'product_uom_qty': 3.0,
                 'product_id': self.get_odoo_data().id},
            ]
            sale_order = cls.env['sale.order'].create({
                'name': '销售订单1',
                'partner_id': cls.env['res.partner'].search([])[0],
                'order_line': [(0, 0, order_line_value) for order_line_value in order_line_values]
            })
            return sale_order
```

    # 多对一创建：[(0, 0, {value})]
    # 多对多创建：[6, 0, [ids]]

```python
    if __name__ == '__main__':
        odoo_data = GetOdooData()
        sale_order = odoo_data.create_sale_order()
```
