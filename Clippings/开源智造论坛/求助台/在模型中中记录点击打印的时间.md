# 在模型中中记录点击打印的时间

> [!info] 来源
> 开源智造论坛 · 求助台 | 作者:姚代娣 | 2025-06-11
> <http://www.thinkltd.cn/forum/1/4049>

需要在各个模型增加时间类型字段，然后重写打印的方法。如下：

```python
from odoo import models, fields, _
from datetime import datetime
from odoo.exceptions import RedirectWarning
import logging
```

_logger = logging.getLogger(__name__)

```python
class IrActionsReport(models.Model):
_inherit = 'ir.actions.report'

def _render_qweb_pdf(self, report_ref, res_ids=None, data=None):
```

# 调用原生方法生成 PDF 内容
pdf_content, report_type = super()._render_qweb_pdf(report_ref, res_ids=res_ids, data=data)

# 如果不是 PDF 类型，跳过后续处理
if report_type != 'pdf':
return pdf_content, report_type

# 获取报表记录（sudo 保证权限）
report_sudo = self._get_report(report_ref)

# 更新打印时间字段
if report_sudo and res_ids:
model = report_sudo.model
records = self.env[model].browse(res_ids)

try:
if model == 'mrp.production':
records.write({'last_printed_at': fields.Datetime.now()})
elif model == 'stock.picking':
records.write({'last_printed_at': fields.Datetime.now()})
elif model == 'sale.order':
records.write({'last_printed_at': fields.Datetime.now()})
elif model == 'purchase.order':
records.write({'last_printed_at': fields.Datetime.now()})

_logger.info("The PDF report has been generated for model: %s, records %s.", model, str(res_ids))
except Exception as e:
_logger.warning("Failed to update last_printed_at for model %s: %s", model, str(e))

return pdf_content, report_type
